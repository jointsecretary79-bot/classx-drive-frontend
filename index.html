// Free email-OTP routes for the Class X Drive backend (Express on Render). // npm i express cors nodemailer // Render env vars: SMTP_HOST, SMTP_PORT, SMTP_USER, SMTP_PASS, MAIL_FROM, OTP_SECRET // Gmail example: SMTP_HOST=smtp.gmail.com SMTP_PORT=465 SMTP_USER=you@gmail.com // SMTP_PASS= (Google Account -> Security -> 2-Step Verification -> App passwords) // In your server: app.use(cors()); app.use(express.json()); require('./otp-backend')(app); const crypto = require('crypto'); const nodemailer = require('nodemailer'); const OTP_TTL_MS = 5 * 60 * 1000; const RESEND_MS = 30 * 1000; const MAX_ATTEMPTS = 5; const store = new Map(); // email -> { hash, expires, attempts, lastSent } const transporter = nodemailer.createTransport({ host: process.env.SMTP_HOST, port: Number(process.env.SMTP_PORT || 587), secure: Number(process.env.SMTP_PORT) === 465, auth: { user: process.env.SMTP_USER, pass: process.env.SMTP_PASS }, }); const hash = (email, code) => crypto.createHash('sha256').update(`${email}:${code}:${process.env.OTP_SECRET || 'change-me'}`).digest('hex'); const validEmail = e => typeof e === 'string' && /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(e) && e.length < 254; module.exports = function mountOtp(app) { app.post('/api/otp/send', async (req, res) => { const email = String(req.body?.email || '').trim().toLowerCase(); if (!validEmail(email)) return res.status(400).json({ error: 'Enter a valid email address.' }); const prev = store.get(email); if (prev && Date.now() - prev.lastSent < RESEND_MS) return res.status(429).json({ error: 'Please wait a few seconds before requesting another OTP.' }); const code = String(crypto.randomInt(0, 1000000)).padStart(6, '0'); store.set(email, { hash: hash(email, code), expires: Date.now() + OTP_TTL_MS, attempts: 0, lastSent: Date.now() }); try { await transporter.sendMail({ from: process.env.MAIL_FROM || process.env.SMTP_USER, to: email, subject: 'Your Class X Drive sign-in code', text: `Your OTP is ${code}. It expires in 5 minutes. If you didn't request it, ignore this email.`, }); res.json({ ok: true }); } catch (err) { console.error('OTP mail failed:', err.message); store.delete(email); res.status(500).json({ error: 'Could not send the email. Try again later.' }); } }); app.post('/api/otp/verify', (req, res) => { const email = String(req.body?.email || '').trim().toLowerCase(); const code = String(req.body?.code || '').trim(); const rec = store.get(email); if (!rec || Date.now() > rec.expires) { store.delete(email); return res.status(400).json({ error: 'OTP expired. Request a new one.' }); } if (rec.attempts >= MAX_ATTEMPTS) { store.delete(email); return res.status(429).json({ error: 'Too many attempts. Request a new OTP.' }); } rec.attempts++; const a = Buffer.from(rec.hash), b = Buffer.from(hash(email, code)); if (!crypto.timingSafeEqual(a, b)) return res.status(400).json({ error: 'Wrong OTP. Try again.' }); store.delete(email); res.json({ ok: true, name: email.split('@')[0] }); }); };
